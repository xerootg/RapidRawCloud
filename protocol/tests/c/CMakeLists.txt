# gcc/ctest conformance test for the generated C SDK:
#   cmake -S protocol/tests/c -B build && cmake --build build && ctest --test-dir build --output-on-failure
cmake_minimum_required(VERSION 3.16)
project(rrcloud_proto_c_conformance C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -O2 -g)
set(GEN ${CMAKE_CURRENT_SOURCE_DIR}/../../gen/c)
set(B3 ${CMAKE_CURRENT_SOURCE_DIR}/../../../firmware/rrcloud-ingest/components/rrc_blake3)
add_library(rrcloud_proto STATIC ${GEN}/rrcloud_proto.c)
target_include_directories(rrcloud_proto PUBLIC ${GEN})
target_compile_options(rrcloud_proto PRIVATE -Wpedantic -Werror)
add_executable(test_conformance test_conformance.c ${B3}/rrc_blake3.c)
target_include_directories(test_conformance PRIVATE ${B3}/include)
target_link_libraries(test_conformance rrcloud_proto)
add_executable(test_cpp test_cpp.cpp ${B3}/rrc_blake3.c)
target_include_directories(test_cpp PRIVATE ${B3}/include)
target_link_libraries(test_cpp rrcloud_proto)
target_compile_options(test_cpp PRIVATE -Werror)
enable_testing()
add_test(NAME c_conformance COMMAND test_conformance ${CMAKE_CURRENT_SOURCE_DIR}/../../fixtures)
add_test(NAME cpp_wrapper COMMAND test_cpp ${CMAKE_CURRENT_SOURCE_DIR}/../../fixtures)
